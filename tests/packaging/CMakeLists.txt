cmake_minimum_required(VERSION 3.20)
project(RTIMcapConsumer LANGUAGES CXX)

set(RTI_MCAP_COMPONENT core CACHE STRING "Component to exercise")
set(RTI_MCAP_SUBPROJECT_SOURCE "" CACHE PATH "Source checkout for subproject consumption")
if(RTI_MCAP_SUBPROJECT_SOURCE)
  set(CMAKE_BUILD_TYPE Release)
  set(BUILD_SHARED_LIBS OFF)
  set(BUILD_TESTING ON)
  add_subdirectory("${RTI_MCAP_SUBPROJECT_SOURCE}" rti-mcap)
  if(RTI_MCAP_BUILD_STORAGE OR RTI_MCAP_BUILD_TESTS OR RTI_MCAP_BUILD_BENCHMARKS)
    message(FATAL_ERROR "Subproject unexpectedly enabled storage, tests, or benchmarks")
  endif()
  if(BUILD_SHARED_LIBS OR NOT BUILD_TESTING OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Subproject changed parent build preferences")
  endif()
else()
  if(RTI_MCAP_COMPONENT STREQUAL "all")
    find_package(RTIMcap 1 CONFIG REQUIRED)
  else()
    find_package(RTIMcap 1 CONFIG REQUIRED COMPONENTS "${RTI_MCAP_COMPONENT}")
  endif()
endif()

if(RTI_MCAP_COMPONENT STREQUAL "storage" OR
    (RTI_MCAP_COMPONENT STREQUAL "all" AND TARGET rti::mcap::storage::record))
  foreach(plugin IN ITEMS record replay)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${plugin}-$<CONFIG>.txt"
      CONTENT "$<TARGET_FILE:rti::mcap::storage::${plugin}>\n")
  endforeach()
else()
  if(NOT RTI_MCAP_SUBPROJECT_SOURCE AND TARGET rti::mcap::storage::record)
    message(FATAL_ERROR "SDK component unexpectedly loaded storage targets")
  endif()
  set(component "${RTI_MCAP_COMPONENT}")
  if(component STREQUAL "all")
    set(component file)
  endif()
  add_executable(consumer "${component}.cxx")
  target_link_libraries(consumer PRIVATE rti::mcap::${component})
endif()
